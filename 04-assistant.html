lesson1_sft.py
import math, random
random.seed(0)

def train(corpus, W=None, epochs=60):          # 第一册的训练循环,原封不动
    global vocab, idx, n
    vocab = sorted(set(corpus)); n = len(vocab)
    idx = {w: i for i, w in enumerate(vocab)}
    W = W or [[random.uniform(-.1, .1) for _ in range(n)] for _ in range(n)]
    for _ in range(epochs):
        for a, b in zip(corpus[:-1], corpus[1:]):
            ex = [math.exp(v) for v in W[idx[a]]]; s = sum(ex)
            for j in range(n):
                W[idx[a]][j] -= .5 * (ex[j]/s - (1 if j == idx[b] else 0))
    return W

def ask(W, prompt, steps=4):
    w, out = prompt[-1], list(prompt)
    for _ in range(steps):
        p = W[idx[w]]
        w = vocab[p.index(max(p))]; out.append(w)
    return " ".join(out)

# 阶段①预训练:语料里"问题后面跟着的是另一个问题"(像试卷)
base = train("问 黑洞 是 什么 ? 问 恒星 是 什么 ? 问 光年 是 什么 ?".split())
print("base 模型 :", ask(base, ["黑洞", "是", "什么", "?"]))

# 阶段②SFT:继续用同一循环,但喂"问题→回答"格式的示范
sft = train("黑洞 是 什么 ? 答 : 引力 极强 的 天体".split(), W=None)
print("SFT 之后  :", ask(sft, ["黑洞", "是", "什么", "?"], steps=6))
# 同一套公式,喂不同数据 → 从"反问你"变成"回答你"。这就是微调的全部。

    公 式
    
top-p 的规则,一句话

    

      

        把词按概率从大到小排,依次累加,加到刚超过 p 就截断,只在留下的词里掷骰子
      

      
把鼠标移到规则的某一部分上试试 →

      
和温度的分工:T 改变骰子的偏心程度(第一册第一课:分数先 ÷T 再 softmax),top-p 决定谁有资格上骰子。T=0 等价于"永远挑最大"(贪心)——同一个问题每次答案完全一样。生成何时停?词表里有个特殊 token(如 <|endoftext|>),模型学会在该结束时给它高概率——"闭嘴"也是预测出来的。

    

  


  

    手算验证
    
口算一次 top-p 截断

    

      四个候选词概率:0.50、0.30、0.15、0.05,top-p = 0.9:

      0.50 → 0.50+0.30=0.80 → 0.80+0.15=0.95 ≥ 0.9,停 —— 前三个词入围,最后那个 5% 的词被踢出。
      
✍️ 轮到你:概率 0.6、0.2、0.1、0.1,top-p=0.85,几个词入围?

    

  


  

    回到大模型
    
API 文档里的 temperature=0.7, top_p=0.9 就是这两个旋钮。写代码、做数学 → 调低 T(要确定性);头脑风暴、写诗 → 调高 T(要多样性)。"同一个问题问两遍答案不同"不是 bug,是骰子在工作。

lesson2_sampling.py

      
import math, random
random.seed(42)

words  = ["好", "不错", "冷", "热", "刺激", "猫"]
logits = [2.0, 1.2, 0.8, 0.3, -0.5, -2.0]      # 模型刚算完的分数

def softmax(z, T=1.0):
    ex = [math.exp(v / T) for v in z]; s = sum(ex)
    return [e / s for e in ex]

def top_p_sample(T=1.0, top_p=0.9):
    p = softmax(logits, T)
    ranked = sorted(zip(words, p), key=lambda x: -x[1])
    kept, acc = [], 0.0
    for w, q in ranked:                        # 累加到刚超过 top_p 就截断
        kept.append((w, q)); acc += q
        if acc >= top_p: break
    r, acc = random.random() * sum(q for _, q in kept), 0.0
    for w, q in kept:
        acc += q
        if r <= acc: return w
    return kept[-1][0]

for T in [0.3, 1.0, 2.0]:
    print(f"T={T}   15 次采样:", " ".join(top_p_sample(T) for _ in range(15)))
# T=0.3 几乎全是"好";T=2.0 五花八门——你的 ChatGPT 温度旋钮,就是它。

上下文窗口与 KV 缓存:长对话的物理极限
context window · KV cache

  
两件事决定了聊天体验的"物理规律":KV 缓存让逐词生成快得起来,上下文窗口决定模型最多能"看见"多少字。

⚡ 缓存每个词的 K/V

      

      
原理:第一册第五课里,每个词的 K 和 V 算好之后永远不变——那就存起来!新词只需算自己的 Q/K/V,再和缓存里的所有 K 做点积。代价:缓存占显存,这就是长上下文昂贵的原因。

"128k 上下文"就是窗口大小(单位:token,第二册第一课)。长对话里模型"忘了开头"不是记性差,是开头物理上不在输入里了(或被摘要压缩过)。窗口内的内容也不是平等的——太长时中间部分的注意力常被稀释("lost in the middle")。每次新请求,模型都是从零读一遍窗口内的全部内容:它没有"记忆",只有"眼前这一屏"。

lesson3_kvcache.py

      
# 生成第 t 个词时,Attention 要做 t 次"Q·K 点积"
# 不缓存:每一步都重算前面所有词的 K/V → 总点积数 = 1+2+...+n
# 缓 存:老词的 K/V 存着不动,每步只算新增的一行

for n in [10, 100, 1000, 10000]:
    no_cache = n * (n + 1) * (n + 2) // 6      # 每步重算全部历史(累计)
    cached   = n * (n + 1) // 2                # 每步只加一行
    print(f"生成 {n:>6} 词   不缓存 ≈ {no_cache:>16,}   缓存 ≈ {cached:>12,}   省 {no_cache//cached} 倍")
# 生成一万词时差一万倍——没有 KV 缓存,就没有流畅的 ChatGPT。

import math, random
random.seed(7)

# ── 先用第一册的方法,训练同一个玩具模型 ─────────────────
corpus = ("今天 天气 很 好 。 今天 天气 很 冷 。 明天 天气 很 好 。 "
          "今天 心情 很 好 。 明天 心情 很 不错 。").split()
vocab = sorted(set(corpus)); n = len(vocab)
idx = {w: i for i, w in enumerate(vocab)}
W = [[random.uniform(-0.1, 0.1) for _ in range(n)] for _ in range(n)]

def softmax(z, T=1.0):
    ex = [math.exp(v / T) for v in z]; s = sum(ex)
    return [e / s for e in ex]

for _ in range(80):
    for prev, nxt in zip(corpus[:-1], corpus[1:]):
        p = softmax(W[idx[prev]])
        for j in range(n):
            W[idx[prev]][j] -= 0.5 * (p[j] - (1 if j == idx[nxt] else 0))

# ── 本册新增:带温度 + top-p 的采样器 ────────────────────
def sample(w, T=1.0, top_p=0.95):
    p = softmax(W[idx[w]], T)
    ranked = sorted(zip(vocab, p), key=lambda x: -x[1])
    kept, acc = [], 0.0
    for tok, q in ranked:                      # 概率从大到小累加,超过 top_p 截断
        kept.append((tok, q)); acc += q
        if acc >= top_p: break
    r, acc = random.random() * sum(q for _, q in kept), 0.0
    for tok, q in kept:
        acc += q
        if r <= acc: return tok
    return kept[-1][0]

def generate(T):
    w, sent = "今天", ["今天"]
    for _ in range(6):
        w = sample(w, T)
        sent.append(w)
        if w == "。": break                     # 「。」就是它的"结束 token"
    return " ".join(sent)

for T in [0.2, 1.0, 3.0]:
    print(f"T={T}")
    for _ in range(3):
        print("  ", generate(T))

    

  

  
观察输出:T=0.2 三次几乎一模一样(复读机);T=1.0 在几个合理句子间切换;T=3.0 开始胡言乱语——语法崩坏的样子,和真实大模型把温度拉满时如出一辙。再试试把 top_p 改成 0.5,高温也会被"资格审查"拉回正轨。
